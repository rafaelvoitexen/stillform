<script is:inline>
  (() => {
    const hero = document.getElementById('top');
    const lease = window.__stillformHeroPaint;
    if (!hero || !lease) return;
    performance.mark('stillform:hero:candidate');
    requestAnimationFrame(() => { lease.open = false; performance.mark('stillform:hero:deadline'); });
    const skip = reason => {
      hero.dataset.entrance = 'skipped';
      hero.dataset.entranceReason = reason;
    };
    const reason = () => {
      if (!lease.open || performance.getEntriesByType('paint').length) return 'already-painted';
      if (document.visibilityState !== 'visible') return 'background';
      if (matchMedia('(prefers-reduced-motion: reduce)').matches) return 'reduced-motion';
      if (scrollY > 1 || (location.hash && location.hash !== '#top')) return 'away-from-top';
      if (performance.getEntriesByType('navigation')[0]?.type === 'back_forward') return 'history';
      return null;
    };
    const earlyReason = reason();
    if (earlyReason) return skip(earlyReason);
    const image = hero.querySelector('img');
    // Assets may finish in this pre-paint opportunity; content remains fully visible while they load.
    Promise.all([lease.font, lease.decoded]).then(([, decoded]) => {
      if (!decoded || !image.complete || !image.naturalWidth) return skip('image-not-ready');
      const lateReason = reason();
      if (lateReason) return skip(lateReason);
      if (!document.fonts.check('500 176px Onest') ||
          ![...document.fonts].some(face => face.family === 'Onest' && face.status === 'loaded')) return skip('font-not-ready');
      const lines = [...hero.querySelectorAll('.line-motion')];
      if (getComputedStyle(hero).display !== 'flex' || lines.some(line => getComputedStyle(line).display !== 'block')) return skip('styles-not-ready');
      if (lines.some(line => line.getBoundingClientRect().height > parseFloat(getComputedStyle(line).lineHeight) + 1)) return skip('text-reflow');
      // Admission happens before paint. A late module may only adopt this already-admitted state.
      hero.dataset.entrance = 'prepared';
      performance.mark('stillform:hero:prepared');
      const preparationEvents = new AbortController();
      lease.release = () => { clearTimeout(lease.deadline); preparationEvents.abort(); };
      const interrupt = () => {
        if (hero.dataset.entrance === 'prepared') skip('preparation-interrupted');
        lease.release();
      };
      const options = { signal: preparationEvents.signal };
      window.addEventListener('scroll', interrupt, { ...options, passive: true });
      window.addEventListener('resize', interrupt, options);
      window.addEventListener('pagehide', interrupt, options);
      hero.addEventListener('focusin', interrupt, options);
      hero.addEventListener('click', interrupt, options);
      document.addEventListener('visibilitychange', interrupt, options);
      matchMedia('(prefers-reduced-motion: reduce)').addEventListener('change', interrupt, options);
      lease.deadline = setTimeout(() => {
        if (hero.dataset.entrance === 'prepared') skip('runtime-not-ready');
        lease.release();
      }, 160);
      hero.dispatchEvent(new Event('stillform:hero-ready'));
    }).catch(() => skip('assets-unavailable'));
  })();
</script>
